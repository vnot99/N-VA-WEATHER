<script>
const $ = id => document.getElementById(id);

function weatherInfo(code) {
  if (code === 0) return ["☀️", "Trời quang"];
  if (code <= 2) return ["🌤️", "Ít mây"];
  if (code === 3) return ["☁️", "Nhiều mây"];
  if (code === 45 || code === 48)
    return ["🌫️", "Sương mù"];
  if (code <= 57) return ["🌦️", "Mưa phùn"];
  if (code <= 67) return ["🌧️", "Trời mưa"];
  if (code <= 77) return ["❄️", "Tuyết"];
  if (code <= 82) return ["🌧️", "Mưa rào"];
  if (code >= 95) return ["⛈️", "Dông"];
  return ["🌥️", "Thời tiết thay đổi"];
}

function round(n) {
  return Math.round(n);
}

async function getJSON(url) {
  const response = await fetch(url);
  if (!response.ok) throw new Error("Lỗi kết nối dữ liệu.");
  return response.json();
}

async function loadWeather() {
  const city = $("city").value.trim();
  if (!city) {
    $("status").textContent = "Hãy nhập tên thành phố.";
    return;
  }

  $("status").className = "";
  $("status").textContent = "⏳ Đang tải thời tiết...";
  $("cityName").textContent = "Đang tìm địa điểm...";
  $("desc").textContent = "Đang cập nhật...";
  $("forecast").textContent = "Đang tải...";
  $("hourly").textContent = "Đang tải...";

  try {
    const geo = await getJSON(
      "https://geocoding-api.open-meteo.com/v1/search?name=" +
      encodeURIComponent(city) +
      "&count=10&language=vi&format=json"
    );

    if (!geo.results || !geo.results.length) {
      throw new Error("Không tìm thấy thành phố này.");
    }

    const p = geo.results.find(x => x.country_code === "VN")
      || geo.results[0];

    const params = new URLSearchParams({
      latitude: p.latitude,
      longitude: p.longitude,
      current: "temperature_2m,relative_humidity_2m,apparent_temperature,wind_speed_10m,weather_code",
      hourly: "temperature_2m,precipitation_probability,weather_code",
      daily: "weather_code,temperature_2m_max,temperature_2m_min",
      forecast_days: "7",
      timezone: "auto"
    });

    const data = await getJSON(
      "https://api.open-meteo.com/v1/forecast?" + params
    );

    const c = data.current;
    const d = data.daily;
    const h = data.hourly;
    const info = weatherInfo(c.weather_code);

    $("cityName").textContent =
      p.name + (p.admin1 ? ", " + p.admin1 : "");

    $("temp").textContent = round(c.temperature_2m) + "°";
    $("desc").textContent = info[0] + " " + info[1];
    $("humidity").textContent = c.relative_humidity_2m + "%";
    $("wind").textContent = c.wind_speed_10m + " km/h";
    $("feels").textContent = round(c.apparent_temperature) + "°";
    $("max").textContent = round(d.temperature_2m_max[0]) + "°C";
    $("min").textContent = round(d.temperature_2m_min[0]) + "°C";

    const start = Math.max(0, h.time.findIndex(t => t >= c.time));

    $("hourly").innerHTML = h.time.slice(start, start + 24)
      .map((time, j) => {
        const i = start + j;
        const wi = weatherInfo(h.weather_code[i]);
        const label = j === 0 ? "Hiện tại" :
          new Date(time).toLocaleTimeString("vi-VN", {
            hour: "2-digit", minute: "2-digit"
          });

        return `<div class="hour">
          <div>${label}</div>
          <div style="font-size:22px">${wi[0]}</div>
          <div><b>${round(h.temperature_2m[i])}°</b></div>
          <div>💧${h.precipitation_probability?.[i] ?? 0}%</div>
        </div>`;
      }).join("");

    $("forecast").innerHTML = d.time.map((day, i) => {
      const date = new Date(day + "T12:00:00");
      const name = i === 0 ? "Hôm nay" :
        date.toLocaleDateString("vi-VN", {
          weekday: "short", day: "2-digit", month: "2-digit"
        });
      const wi = weatherInfo(d.weather_code[i]);

      return `<div class="day">
        <span class="dayname">${name}</span>
        <span>${wi[0]}</span>
        <span class="daytemp">
          <b>${round(d.temperature_2m_max[i])}°</b>
          / ${round(d.temperature_2m_min[i])}°
        </span>
      </div>`;
    }).join("");

    $("status").textContent =
      "✓ Dữ liệu trực tiếp · " + data.timezone;

  } catch (error) {
    $("status").className = "error";
    $("status").textContent = "⚠️ " + error.message;
    $("cityName").textContent = "Không tải được dữ liệu";
    $("desc").textContent =
      "Kiểm tra Internet rồi nhấn Tìm để thử lại.";
  }
}

$("searchForm").addEventListener("submit", function(event) {
  event.preventDefault();
  loadWeather();
});

loadWeather();
</script>

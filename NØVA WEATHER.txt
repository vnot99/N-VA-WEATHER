<script>
async function loadWeather() {
  const city = document.getElementById("city").value.trim();
  if (!city) return;

  document.getElementById("place").textContent = "Đang tìm...";
  document.getElementById("forecast").textContent = "Đang tải...";

  try {
    const geo = await fetch(
      "https://geocoding-api.open-meteo.com/v1/search?name=" +
      encodeURIComponent(city) + "&count=1&language=vi&format=json"
    ).then(r => r.json());

    if (!geo.results || !geo.results.length) {
      throw new Error("Không tìm thấy thành phố!");
    }

    const p = geo.results[0];

    const url = "https://api.open-meteo.com/v1/forecast?" +
      "latitude=" + p.latitude +
      "&longitude=" + p.longitude +
      "&current=temperature_2m,relative_humidity_2m,apparent_temperature,wind_speed_10m,weather_code" +
      "&daily=weather_code,temperature_2m_max,temperature_2m_min" +
      "&forecast_days=7&timezone=auto";

    const data = await fetch(url).then(r => r.json());
    const c = data.current;

    const weather = {
      0: "☀️ Trời quang",
      1: "🌤️ Ít mây",
      2: "⛅ Có mây",
      3: "☁️ Nhiều mây",
      45: "🌫️ Sương mù",
      48: "🌫️ Sương mù",
      51: "🌦️ Mưa phùn",
      53: "🌦️ Mưa phùn",
      55: "🌧️ Mưa phùn",
      61: "🌧️ Mưa nhẹ",
      63: "🌧️ Mưa vừa",
      65: "🌧️ Mưa lớn",
      80: "🌦️ Mưa rào",
      81: "🌧️ Mưa rào",
      82: "⛈️ Mưa rào lớn",
      95: "⛈️ Dông"
    };

    document.getElementById("place").textContent =
      p.name + (p.country ? ", " + p.country : "");

    document.getElementById("temp").textContent =
      Math.round(c.temperature_2m) + "°";

    document.getElementById("desc").textContent =
      weather[c.weather_code] || "🌡️ Thời tiết thay đổi";

    document.getElementById("details").textContent =
      "💧 Độ ẩm: " + c.relative_humidity_2m + "%" +
      " | 💨 Gió: " + c.wind_speed_10m + " km/h" +
      " | 🌡️ Cảm giác: " +
      Math.round(c.apparent_temperature) + "°";

    document.getElementById("forecast").innerHTML =
      data.daily.time.map((day, i) => {
        const d = new Date(day + "T12:00:00");
        const name = d.toLocaleDateString("vi-VN", {
          weekday: "long", day: "2-digit", month: "2-digit"
        });

        return `<div class="card">
          📆 ${name}<br>
          🌡️ Cao ${Math.round(data.daily.temperature_2m_max[i])}°
          · Thấp ${Math.round(data.daily.temperature_2m_min[i])}°
        </div>`;
      }).join("");

  } catch (e) {
    document.getElementById("place").textContent = "Có lỗi xảy ra";
    document.getElementById("desc").textContent = e.message;
    document.getElementById("forecast").textContent =
      "Hãy kiểm tra Internet và thử lại.";
  }
}

loadWeather();
</script>
